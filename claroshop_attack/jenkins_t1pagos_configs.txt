def nexusArtifactLifeCycleManager(String nexusRepoURI, String nexusManagedArtifactName) {
    
    KEEP_N_ARTIFACTS=&quot;5&quot;
    withCredentials([usernameColonPassword(credentialsId: &apos;nexus&apos;, variable: &apos;NX_PWD1&apos;)]) {
        sh &quot;&quot;&quot;#!/bin/bash +x 
                curl 2&gt;/dev/null -f -u ${NX_PWD1} ${NEXUS_REPO_URI}/${JOB_NAME}/deployed.txt -o deployed.txt || true
                curl -s -f -u ${NX_PWD1} --upload-file ${nexusManagedArtifactName}.tar.gz ${NEXUS_REPO_URI}/${JOB_NAME}/${BUILD_NUMBER}/${nexusManagedArtifactName}.tar.gz &amp;&amp; 
                echo ${BUILD_NUMBER}/${nexusManagedArtifactName}.tar.gz &gt;&gt; deployed.txt &amp;&amp;
                echo &quot;Mark files for removal...&quot;
                egrep &quot;[0-9]+/${nexusManagedArtifactName}&quot; deployed.txt | grep -v permanent | head -n-${KEEP_N_ARTIFACTS} &gt; mark4delete.txt || true
                cat mark4delete.txt
                echo &quot;Index preserved files...&quot;
                egrep -v &quot;[0-9]+/${nexusManagedArtifactName}&quot; deployed.txt  &gt; preserved.txt || true
                egrep &quot;[0-9]+/${nexusManagedArtifactName}&quot; deployed.txt | grep -v permanent | tail -n${KEEP_N_ARTIFACTS} &gt;&gt; preserved.txt || true 
                cat preserved.txt
                cat mark4delete.txt | while read file; do 
                    echo &quot;Delete \$file&quot;
                    curl -s -u ${NX_PWD1} -X DELETE ${NEXUS_REPO_URI}/${JOB_NAME}/\$file
                done
                echo &quot;Upload files index&quot;
                curl -s -f -u ${NX_PWD1} --upload-file preserved.txt ${NEXUS_REPO_URI}/${JOB_NAME}/deployed.txt
                &quot;&quot;&quot;            
    }

}
