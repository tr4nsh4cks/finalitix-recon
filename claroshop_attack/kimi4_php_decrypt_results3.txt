     * @param int
     * 
     */
    public function guardaTarjeta($idUsuario,$nombre,$apellido,Card $tarjeta,$idDir)
    {
        
        //setea las variables
        $this->exchangeArray($idUsuario,$nombre,$apellido,$tarjeta,$idDir);

        //valida tamaño y numericos en numero de tarjet, mes y año
        $validacion = $this->validaTarjeta();
        if( $validacion != '' ){
            echo $validacion;
            return false;
        }


        //evita xss y sql injection
        $this->nombre   = htmlspecialchars($this->nombre);
        $this->apellido = htmlspecialchars($this->apellido);
        $this->encrypt();
        //guarda en bd
        $modeloCardOneClick= new  Modelos_CardOneclick();
        
        $error = $modeloCardOneClick->guardaCard($idUsuario,$this->nombre,$this->apellido,$tarjeta,$this->idDir);

    /**
     * Encripta por Rc4
     * @param string
     * 
     */
    private function Rc4Encrypt($string)
    {
        $rc4crypt = new Rc4();
        return $rc4crypt->StringToHexString($rc4crypt->Salaa($string,$this->key));
    }
    /**
     * Desencripta por Rc4
     * @param string
     * 
     */
    private function Rc4Decrypt($string)
    {
        $rc4crypt = new Rc4();
        return $rc4crypt->Salaa($rc4crypt->HexStringToString($string),$this->key);
    }
    /**
     * Valida que los datos de la tarjeta, nombre y apellidos
     * se encuentren como se requieren
     * 
     */
    public function validaTarjeta()
    {
        $error  = '';
        $error .= $this->valFunctions('Numero de tarjeta',$this->tarjeta->getAccountNumber(),16);
        $error .= $this->valFunctions('Mes de expiracion',$this->tarjeta->getExpirationMonth(),2);
        $error .= $this->valFunctions('Año de expiracion',$this->tarjeta->getExpirationYear(),2);

        return $error;
    }
    /**
     * Funciones para validar
     * 
     */
    private function valFunctions($tipo,$dato,$length)
    {
        $res='';
        if( strlen($dato)!=$length ){
            $res .=  $tipo." no es del tamanio ".$length.", mide ".strlen($dato)." : ".$dato."<br/>";
        }
        
        $dato = intval($dato)!=0?intval($dato):"";
        if( !is_int($dato) ){
            $res .=  $tipo." no es numerico ".$dato."<br/>";
        }
        
        return $res;
    }

    protected function runTransaction ()
    {

        $context_options = array(
            'http' => array(
                'timeout' => $this->timeout,
            ),
        );
        $context = stream_context_create( $context_options );
        // options we pass into the soap client
        $soap_options = array(
            'compression' => SOAP_COMPRESSION_ACCEPT | SOAP_COMPRESSION_GZIP | SOAP_COMPRESSION_DEFLATE,// turn on HTTP compression
            'encoding' => 'utf-8',// set the internal character encoding to avoid random conversions
            'exceptions' => true,// throw SoapFault exceptions when there is an error
            'connection_timeout' => $this->timeout,
            'stream_context' => $context,
        );
        $soap_options['cache_wsdl'] = ( $this->environment == "prod" ) ? WSDL_CACHE_BOTH : WSDL_CACHE_NONE;
        try {           
            $soap = new \SoapClient($this->wsdl, $soap_options );
            $this->addWsseToken($soap);
            $this->createRequest();
            //error_log(print_r($this->request, true));
            $response = $soap->runTransaction($this->request); 
            $this->response = $response;
        } catch ( Exception $ex ) {

            error_log( $ex->getMessage());
        }
    }
    private function addWsseToken($soap)
    {
        $wsse_namespace = 'http://docs.oasis-open.org/wss/2004/01/oasis-200401-wss-wssecurity-secext-1.0.xsd';
        $type_namespace = 'http://docs.oasis-open.org/wss/2004/01/oasis-200401-wss-username-token-profile-1.0#PasswordText';
        $user = new \SoapVar( $this->merchantID, XSD_STRING, null, $wsse_namespace, null, $wsse_namespace );
        $pass = new \SoapVar( $this->transaction_key, XSD_STRING, null, $type_namespace, null, $wsse_namespace );
        // create the username token container object
        $username_token = new \stdClass();
        $username_token->Username = $user;
        $username_token->Password = $pass;
        // convert the username token object into a soap var
        $username2_token = new \SoapVar( $username_token, SOAP_ENC_OBJECT, null, $wsse_namespace, 'UsernameToken', $wsse_namespace );
        // create the security container object
        $security = new \stdClass();
        $security->UsernameToken = $username2_token;
        // convert the security container object into a soap var
        $securitySoap = new \SoapVar( $security, SOAP_ENC_OBJECT, null, $wsse_namespace, 'Security', $wsse_namespace );
        // create the header out of the security soap var
        $header = new \SoapHeader( $wsse_namespace, 'Security', $securitySoap, true );
        // add the headers to the soap client
        $soap->__setSoapHeaders( $header );
    }
